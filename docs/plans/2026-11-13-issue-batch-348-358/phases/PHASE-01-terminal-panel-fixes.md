- Unicode 11 widths cross the real xterm and the real addon (Task 1.10).

  - `export interface TerminalSizeDiagnosticRecord { readonly at: number; readonly trigger: TerminalSizeDiagnosticTrigger; readonly rendererGeneration: number; readonly mountWidth: number; readonly mountHeight: number; readonly fitted: TerminalDimensions | null; readonly applied: TerminalDimensions | null; readonly role: "owner" | "mirror" | "legacy"; readonly devicePixelRatio: number; readonly fontStatus: "loading" | "loaded" | "unavailable"; readonly renderer: "webgl" | "standard"; readonly request: { readonly cols: number; readonly rows: number; readonly outcome: TerminalResizeOutcome | "pending" | "failed" } | null; readonly repaint: "none" | "scheduled" | "done"; }`
